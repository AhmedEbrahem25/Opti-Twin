## RESPONSIBILITY
Configure GSAP, define custom eases, implement the universal scroll reveal system,  
hero title animation (SplitText), and Anime.js SVG/counter utilities.

## SCOPE BOUNDARIES
- ✅ GSAP global config + plugin registration
- ✅ Custom ease definitions
- ✅ `initScrollReveal()` — universal `data-reveal` system
- ✅ `animateHeroTitle()` — SplitText 3D char entrance
- ✅ Anime.js: SVG path draw, counter odometer, stagger grid
- ❌ Does NOT handle scroll physics (→ SCROLL module — Lenis)
- ❌ Does NOT handle page transitions (→ SCROLL module — Barba)
- ❌ Does NOT handle intro sequence timing (→ INTRO module)
- ❌ Does NOT handle 3D render loop (→ WEBGL module)

## WHEN TO USE
Every project. GSAP is the animation foundation.

## WHEN NOT TO USE
Never skip GSAP setup. You may skip Anime.js utilities if no SVG/counters are needed.

## CONFLICTS
- Never run a `requestAnimationFrame` loop outside the GSAP ticker.
- Do not use CSS `animation` or `transition` for elements that GSAP also animates.

## GSAP GLOBAL SETUP

```javascript
// ═══════════════════════════════════════════════
//  scripts/main.js — ALWAYS FIRST in DOMContentLoaded
// ═══════════════════════════════════════════════

// 1. Config — suppress console warnings in production
gsap.config({ nullTargetWarn: false, trialWarn: false });

// 2. Register plugins — before any usage
gsap.registerPlugin(ScrollTrigger, SplitText, CustomEase);

// 3. Sensible defaults — overridden per animation as needed
gsap.defaults({ ease: 'power3.out', duration: 0.8 });

// 4. Custom eases — register once, reference by name everywhere
CustomEase.create('cinematic', '0.25, 0.46, 0.45, 0.94');
CustomEase.create('overshoot', '0.34, 1.2,  0.64, 1');
CustomEase.create('snappy',    '0.23, 1,    0.32, 1');
```

### Named Ease Reference

| Name | Cubic Bezier | Best for |
|---|---|---|
| `cinematic` | `0.25, 0.46, 0.45, 0.94` | Hero reveals, section entrances |
| `overshoot` | `0.34, 1.2, 0.64, 1` | Buttons, cards, interactive elements |
| `snappy` | `0.23, 1, 0.32, 1` | UI state changes, hover responses |
| `power3.out` | GSAP built-in | General-purpose smooth deceleration |
| `power4.out` | GSAP built-in | Fast-start slow-finish — type reveals |
| `power3.inOut` | GSAP built-in | Page transitions, modal opens |

## SPLITTEXT — HERO TITLE ENTRANCE

```javascript
// ═══════════════════════════════════════════════
//  scripts/animation.js
// ═══════════════════════════════════════════════

/**
 * animateHeroTitle
 * 3D character-by-character flip entrance.
 * Used for: intro title, hero section h1, any large display text.
 *
 * @param {string} selector — CSS selector for the element
 * @returns {gsap.core.Timeline}
 */
function animateHeroTitle(selector) {
  const el = document.querySelector(selector);
  if (!el) return;

  el.style.overflow = 'hidden';

  const split = new SplitText(el, {
    type: 'chars,words,lines',
    linesClass: 'line-wrap',
  });

  // Line overflow hidden — prevents chars from showing outside wrapper
  split.lines.forEach(line => {
    line.style.overflow = 'hidden';
    line.style.display = 'block';
  });

  return gsap.timeline()
    .from(split.chars, {
      opacity:         0,
      y:               100,
      rotateX:         -80,
      scaleY:          0.8,
      stagger:         { amount: 0.5, from: 'start' },
      duration:        1.1,
      ease:            'power4.out',
      delay:           0.2,
      transformOrigin: '0% 50% -40px',  // Z-offset sells the 3D depth
    });
}
```

> **Key detail:** `transformOrigin: '0% 50% -40px'` pushes the rotation pivot  
> behind the plane — this is what makes it look like a real 3D flip, not a CSS trick.

## SCROLL REVEAL SYSTEM — UNIVERSAL `data-reveal`

Every content element on every page uses this system. It is the backbone of the scroll experience.

```javascript
/**
 * initScrollReveal
 * Registers all [data-reveal] elements with ScrollTrigger.
 * Must be called:
 *   - After initMainExperience()
 *   - After every Barba page transition (in barba.hooks.after)
 */
function initScrollReveal() {

  const revealMap = {
    'fade-up':    { opacity: 0, y: 50,  filter: 'blur(4px)' },
    'fade-in':    { opacity: 0,         filter: 'blur(10px)' },
    'fade-down':  { opacity: 0, y: -40 },
    'slide-left': { opacity: 0, x: -80 },
    'slide-right':{ opacity: 0, x:  80 },
    'scale-up':   { opacity: 0, scale: 0.88, filter: 'blur(4px)' },
    'clip-up':    { clipPath: 'inset(100% 0% 0% 0%)', opacity: 1 },
    'clip-left':  { clipPath: 'inset(0% 100% 0% 0%)', opacity: 1 },
    'rotate-up':  { opacity: 0, rotateX: -30, y: 40, transformPerspective: 600 },
  };

  gsap.utils.toArray('[data-reveal]').forEach((el) => {
    const type     = el.dataset.reveal          || 'fade-up';
    const delay    = parseFloat(el.dataset.revealDelay    || 0);
    const duration = parseFloat(el.dataset.revealDuration || 1.0);
    const stagger  = el.dataset.revealStagger;
    const fromVars = revealMap[type] || revealMap['fade-up'];

    const config = {
      ...fromVars,
      duration,
      ease: 'power3.out',
      delay,
      scrollTrigger: {
        trigger:       el,
        start:         'top 88%',
        toggleActions: 'play none none none',
        once:          true,
      },
    };

    if (stagger) config.stagger = parseFloat(stagger);

    gsap.from(el, config);
  });
}
```

### HTML Usage

```html
<!-- Basic upward fade -->
<div data-reveal="fade-up">Content</div>

<!-- Staggered children -->
<ul data-reveal="fade-up" data-reveal-stagger="0.08">...</ul>

<!-- Clip reveal with delay -->
<h2 data-reveal="clip-up" data-reveal-delay="0.2" data-reveal-duration="0.9">Heading</h2>

<!-- Rotate entrance -->
<p data-reveal="rotate-up" data-reveal-delay="0.3">Paragraph</p>
```

### Reveal Types Reference

| `data-reveal` value | Motion | Best for |
|---|---|---|
| `fade-up` | Opacity + upward + blur | Default — most elements |
| `fade-in` | Opacity + blur only | Images, full-bleed sections |
| `fade-down` | Opacity + downward | Nav items, dropdowns |
| `slide-left` | Opacity + from left | Right-column content |
| `slide-right` | Opacity + from right | Left-column content |
| `scale-up` | Opacity + scale + blur | Cards, feature blocks |
| `clip-up` | Clip-path reveal bottom-to-top | Headlines, bold reveals |
| `clip-left` | Clip-path reveal right-to-left | Horizontal reveals |
| `rotate-up` | 3D rotateX + upward | Premium feel, headings |

## ANIME.JS UTILITIES

```javascript
// ── SVG PATH DRAW-ON ────────────────────────────
/**
 * Draws SVG paths with stroke animation.
 * Use for: logos, icons, decorative linework.
 */
function animateSVGPaths(selector) {
  anime({
    targets: `${selector} path, ${selector} line, ${selector} circle`,
    strokeDashoffset: [anime.setDashoffset, 0],
    easing:    'easeInOutCubic',
    duration:  2400,
    delay:     (el, i) => i * 180,
    direction: 'forward',
  });
}

// ── NUMBER ODOMETER ─────────────────────────────
/**
 * Animates a number from 0 to target with easing.
 * Emil Kowalski style: smooth expo ease, localeSeparators.
 * @param {HTMLElement} el — the element to update
 * @param {number} target — final number
 * @param {string} suffix — e.g. '%', 'K', '+'
 */
function animateCounter(el, target, suffix = '') {
  const obj = { val: 0 };
  anime({
    targets:  obj,
    val:      target,
    easing:   'easeOutExpo',
    round:    1,
    duration: 2200,
    update:   () => { el.textContent = obj.val.toLocaleString() + suffix; },
  });
}

// ── STAGGER GRID ENTRANCE ───────────────────────
/**
 * Staggers a grid of elements from center outward.
 * Use for: card grids, image galleries, feature lists.
 * @param {string} selector — CSS selector for grid items
 * @param {number} cols — number of columns in the grid
 */
function animateGrid(selector, cols = 4) {
  const items = document.querySelectorAll(selector);
  anime({
    targets:      selector,
    opacity:      [0, 1],
    translateY:   [30, 0],
    scale:        [0.96, 1],
    delay:        anime.stagger(60, {
      grid: [cols, Math.ceil(items.length / cols)],
      from: 'center',
    }),
    easing:   'easeOutExpo',
    duration: 700,
  });
}
```

## PARALLAX DEPTH LAYERS

```javascript
/**
 * initParallax
 * Applies scroll-based depth to layered elements.
 * Uses CSS variable --scroll-velocity (set by Lenis).
 *
 * Depth convention:
 *   data-parallax="0.1"  → background (slow)
 *   data-parallax="0.3"  → mid-ground
 *   data-parallax="0.6"  → foreground (faster)
 */
function initParallax() {
  const els = gsap.utils.toArray('[data-parallax]');

  els.forEach(el => {
    const depth = parseFloat(el.dataset.parallax || 0.2);
    gsap.to(el, {
      y: () => -(window.__lenis?.scroll || 0) * depth,
      ease: 'none',
      scrollTrigger: {
        trigger:   document.body,
        start:     'top top',
        end:       'bottom bottom',
        scrub:     true,
      },
    });
  });
}
```

## PERFORMANCE IMPACT
- **Medium** — SplitText wraps each character in a `<div>`, increasing DOM nodes significantly for large headings.
- ScrollTrigger pins use IntersectionObserver internally — performant at scale.
- Always set `once: true` on reveal ScrollTriggers — prevents re-firing.
- Remove `will-change` from animated elements once animation completes:

```javascript
gsap.from(el, {
  // ...
  onComplete: () => { el.style.willChange = 'auto'; }
});
```
