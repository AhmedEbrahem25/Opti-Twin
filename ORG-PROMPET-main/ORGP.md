### Typography Rules (Emil Kowalski standard)
- **Display / Hero font:** Distinctive, editorial — NOT Inter, NOT Roboto, NOT Space Grotesk
  - Serif options: `Playfair Display`, `DM Serif Display`, `Cormorant`, `Bodoni Moda`
  - Sans options: `Syne`, `Clash Display`, `Cabinet Grotesk`, `Satoshi`, `General Sans`, `Neue Haas`
- **Body font:** Clean, refined — `DM Sans`, `Outfit`, `Plus Jakarta Sans`, `Figtree`
- **Mono:** `JetBrains Mono`, `Fira Code`, `Geist Mono`
- **Type scale (modular):** 11 / 13 / 14 / 16 / 20 / 24 / 32 / 48 / 64 / 96 / 128 / 160px
- **Letter spacing:** Display → `-0.03em` to `-0.05em` · Body → `0` · Labels → `0.08em` uppercase
- **Line height:** Display → `0.95–1.05` · Heading → `1.1–1.2` · Body → `1.6–1.75`
- **Font weight:** Max 2 weights per font — typically 300 + 700 or 400 + 600

## 🌊 LENIS — SMOOTH SCROLL (Best-in-class)

```javascript
// ═══════════════════════════════════════════════
//  LENIS INITIALIZATION — Perfect physics config
// ═══════════════════════════════════════════════
function initLenis() {
  const lenis = new Lenis({
    duration: 1.4,
    easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)), // Expo ease
    direction: 'vertical',
    gestureDirection: 'vertical',
    smooth: true,
    smoothTouch: false,
    touchMultiplier: 2,
    infinite: false,
    autoRaf: false, // Manual RAF for sync with Three.js
  });

  // Sync with GSAP — critical for ScrollTrigger accuracy
  lenis.on('scroll', ScrollTrigger.update);
  gsap.ticker.add((time) => { lenis.raf(time * 1000); });
  gsap.ticker.lagSmoothing(0); // Prevent lag accumulation

  // Velocity-driven parallax intensity
  lenis.on('scroll', ({ scroll, velocity, progress }) => {
    document.documentElement.style.setProperty('--scroll-velocity', Math.abs(velocity));
    document.documentElement.style.setProperty('--scroll-progress', progress);
  });

  window.__lenis = lenis; // Global access
  return lenis;
}
```

**Lenis Rules:**
- NEVER `window.scrollY` directly — use `lenis.scroll` or `lenis.on('scroll')`
- NEVER `scroll-behavior: smooth` — Lenis owns scroll
- NEVER `overflow: hidden` on body after init — breaks Lenis
- Velocity value drives parallax depth multiplier
- `autoRaf: false` when Three.js is active — one RAF to rule them all

## 🎬 GSAP ANIMATION SYSTEM

```javascript
// ═══════════════════════════════════════════════
//  GSAP — Global setup (ALWAYS FIRST)
// ═══════════════════════════════════════════════
gsap.config({ nullTargetWarn: false, trialWarn: false });
gsap.registerPlugin(ScrollTrigger, SplitText, CustomEase);
gsap.defaults({ ease: 'power3.out', duration: 0.8 });

// Custom cinematic eases (Emil Kowalski references)
CustomEase.create('cinematic',    '0.25, 0.46, 0.45, 0.94');
CustomEase.create('overshoot',    '0.34, 1.2, 0.64, 1');
CustomEase.create('snappy',       '0.23, 1, 0.32, 1');

// ═══════════════════════════════════════════════
//  HERO TITLE — SplitText cinematic entrance
// ═══════════════════════════════════════════════
function animateHeroTitle(selector) {
  const el = document.querySelector(selector);
  if (!el) return;

  el.style.overflow = 'hidden';
  const split = new SplitText(el, {
    type: 'chars,words,lines',
    linesClass: 'line-wrap',
  });

  split.lines.forEach(line => {
    line.style.overflow = 'hidden';
    line.style.display = 'block';
  });

  return gsap.timeline()
    .from(split.chars, {
      opacity: 0,
      y: 100,
      rotateX: -80,
      scaleY: 0.8,
      stagger: { amount: 0.5, from: 'start' },
      duration: 1.1,
      ease: 'power4.out',
      delay: 0.2,
      transformOrigin: '0% 50% -40px',
    });
}

// ═══════════════════════════════════════════════
//  SCROLL REVEAL — Universal data-reveal system
//  Used on EVERY page, EVERY section, EVERY element
// ═══════════════════════════════════════════════
function initScrollReveal() {
  const revealMap = {
    'fade-up':    { opacity: 0, y: 50, filter: 'blur(4px)' },
    'fade-in':    { opacity: 0, filter: 'blur(10px)' },
    'fade-down':  { opacity: 0, y: -40 },
    'slide-left': { opacity: 0, x: -80 },
    'slide-right':{ opacity: 0, x: 80 },
    'scale-up':   { opacity: 0, scale: 0.88, filter: 'blur(4px)' },
    'clip-up':    { clipPath: 'inset(100% 0% 0% 0%)', opacity: 1 },
    'clip-left':  { clipPath: 'inset(0% 100% 0% 0%)', opacity: 1 },
    'rotate-up':  { opacity: 0, rotateX: -30, y: 40, transformPerspective: 600 },
  };

  gsap.utils.toArray('[data-reveal]').forEach((el) => {
    const type     = el.dataset.reveal || 'fade-up';
    const delay    = parseFloat(el.dataset.revealDelay || 0);
    const duration = parseFloat(el.dataset.revealDuration || 1.0);
    const stagger  = el.dataset.revealStagger;
    const fromVars = revealMap[type] || revealMap['fade-up'];

    const config = {
      ...fromVars,
      duration,
      ease: 'power3.out',
      delay,
      scrollTrigger: {
        trigger: el,
        start: 'top 88%',
        toggleActions: 'play none none none',
        once: true,
      },
    };

    if (stagger) config.stagger = parseFloat(stagger);
    gsap.from(el, config);
  });
}
```

### JavaScript — Intro Sequence (Full 7-Phase)
```javascript
// ═══════════════════════════════════════════════
//  CINEMATIC INTRO — 7-phase orchestrated sequence
// ═══════════════════════════════════════════════

/*
  PHASE OVERVIEW:
  ─────────────────────────────────────────────
  Phase 1 (0.0s):   Ambient background breathes in — scale 1.15→1.0, opacity 0→1
  Phase 2 (0.5s):   Decorative line draws across the screen left→right
  Phase 3 (0.8s):   Eyebrow label fades in — letter-spacing collapses
  Phase 4 (1.0s):   Hero title — each character flips in with 3D rotateX
  Phase 5 (1.9s):   Subtitle fades up through blur
  Phase 6 (2.5s):   CTA button slides up, glows
  Phase 7 (3.2s):   Scroll indicator appears last — site is ready
  ─────────────────────────────────────────────
  Total intro duration: ~3.8s before user interaction
  All phases respect prefers-reduced-motion
*/

function runIntroSequence() {
  // Respect accessibility preference
  const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  if (prefersReduced) {
    // Skip cinematic — instant reveal
    gsap.set('.intro-overlay', { opacity: 1 });
    gsap.set(['.intro-ambient', '.intro-label', '.intro-title', '.intro-subtitle', '.cta-btn', '.scroll-indicator'], {
      opacity: 1, y: 0, filter: 'none', scale: 1, letterSpacing: '0.08em'
    });
    return;
  }

  const tl = gsap.timeline({ defaults: { ease: 'power3.out' } });

  // ── Phase 1: Ambient bg breathes in ──────────
  // The atmosphere arrives first — gives the frame emotional context
  // before any text appears. Scale collapse feels like a camera pulling focus.
  tl.to('.intro-ambient', {
    opacity: 1,
    scale: 1,
    duration: 2.0,
    ease: 'power2.out',
  }, 0);

  // ── Phase 2: Decorative line draws ───────────
  // A single horizontal or geometric line draws itself across the screen.
  // Creates the sense that the space is being "ruled" — measured, deliberate.
  // Use Anime.js for stroke animation precision:
  tl.call(() => {
    const line = document.querySelector('.intro-line');
    if (!line) return;
    gsap.set(line, { opacity: 1 });
    anime({
      targets: '.intro-line line',
      strokeDashoffset: [1440, 0],
      easing: 'easeInOutCubic',
      duration: 900,
    });
  }, [], 0.5);

  // ── Phase 3: Eyebrow label ────────────────────
  // The smallest text comes before the largest.
  // Contrasting scale arrival order creates hierarchy and drama.
  // Letter-spacing collapses: the text "gathers itself" as it arrives.
  tl.to('.intro-label', {
    opacity: 1,
    letterSpacing: '0.08em',
    duration: 0.7,
    ease: 'power2.out',
  }, 0.8);

  // ── Phase 4: Hero title — char-by-char 3D flip ──
  // This is the centrepiece of the intro.
  // Each character enters from below with a 3D perspective flip.
  // The stagger (amount: 0.5) means the word "unfolds" left to right.
  // transformOrigin with z-offset (-40px) sells the depth.
  tl.add(() => animateHeroTitle('.intro-title'), 1.0);

  // ── Phase 5: Subtitle ────────────────────────
  // After the title has fully landed (1.0s + 0.9s = ~1.9s), the subtitle
  // softly reveals itself through blur. The blur-fade combo reads as
  // "coming into focus" — cinematic and purposeful.
  tl.from('.intro-subtitle', {
    opacity: 0,
    y: 18,
    filter: 'blur(4px)',
    duration: 0.9,
  }, 1.9);

  // ── Phase 6: CTA button ───────────────────────
  // The gateway into the experience. Appears last in the content stack.
  // Slides up from below — consistent with the upward reveal language.
  // The glow pulse starts on a slight delay to draw attention.
  tl.from('.cta-btn', {
    opacity: 0,
    y: 28,
    duration: 0.7,
  }, 2.5)
  .call(() => {
    // Subtle glow pulse to invite interaction — not aggressive
    gsap.to('.cta-btn', {
      boxShadow: `0 0 32px var(--accent-glow), 0 0 64px var(--accent-glow-lg)`,
      duration: 1.2,
      ease: 'power2.inOut',
      yoyo: true,
      repeat: -1,
    });
  }, [], 3.1);

  // ── Phase 7: Scroll indicator ─────────────────
  // The last element to appear. Signals that the site is loaded and ready.
  // Fades in gently — doesn't compete with the CTA.
  tl.from('.scroll-indicator', {
    opacity: 0,
    y: 10,
    duration: 0.5,
  }, 3.2);

  return tl;
}
```

```javascript
// ═══════════════════════════════════════════════
//  INTERACTIVE BACKGROUND — Mouse-driven glow movement
// ═══════════════════════════════════════════════

/*
  HOW IT WORKS:
  ─────────────────────────────────────────────
  Mouse moves on screen → target position is updated.
  A lerp loop (RAF) smoothly interpolates each glow's current position
  toward the target at different speeds (creates depth parallax).

  Primary glow:   follows mouse at 8% speed  — most prominent, slowest
  Secondary glow: follows mouse at 12% speed — slightly faster
  Tertiary glow:  follows mouse at 20% speed — fastest, most subtle

  On mobile (pointer: coarse): all mouse tracking is disabled.
  On scroll: glows shift slightly on the Y axis.

  Performance: lerp is extremely cheap. No resize observer needed.
  The glows are fixed-position, so scroll doesn't need to move them
  (scroll influence is applied as an additional Y offset only).
*/

function initInteractiveBackground() {
  const isMobile = !window.matchMedia('(pointer: fine)').matches;
  if (isMobile) return; // Static background on touch devices

  const glows = {
    primary:   {
      el: document.querySelector('.bg-glow--primary'),
      x: 0, y: 0,
      targetX: 0, targetY: 0,
      speed: 0.08,           // 8% — heaviest, slowest
      baseX: 20, baseY: 10, // % of viewport — rest position
    },
    secondary: {
      el: document.querySelector('.bg-glow--secondary'),
      x: 0, y: 0,
      targetX: 0, targetY: 0,
      speed: 0.12,           // 12% — medium
      baseX: 75, baseY: 60,
    },
    tertiary:  {
      el: document.querySelector('.bg-glow--tertiary'),
      x: 0, y: 0,
      targetX: 0, targetY: 0,
      speed: 0.20,           // 20% — lightest, fastest
      baseX: 50, baseY: 70,
    },
  };

  // Mouse position as normalized -0.5→0.5 from center of screen
  let mouse = { x: 0, y: 0 };

  window.addEventListener('mousemove', (e) => {
    mouse.x = (e.clientX / window.innerWidth)  - 0.5;
    mouse.y = (e.clientY / window.innerHeight) - 0.5;

    // Update targets — mouse influence: ±15vw / ±10vh
    Object.values(glows).forEach(glow => {
      glow.targetX = mouse.x * 15; // vw units
      glow.targetY = mouse.y * 10; // vh units
    });
  }, { passive: true });

  // Scroll influence — slight vertical drift
  let scrollInfluence = 0;
  window.__lenis?.on('scroll', ({ scroll }) => {
    // As user scrolls down, glows drift upward slightly
    scrollInfluence = (scroll / document.body.scrollHeight) * -8; // vh
  });

  // ─── LERP LOOP ──────────────────────────────
  // This runs inside the main RAF — NOT a separate loop.
  // Add to GSAP ticker to stay synced with Three.js and Lenis.
  gsap.ticker.add(() => {
    Object.values(glows).forEach(glow => {
      if (!glow.el) return;

      // Linear interpolation: current → target at glow.speed
      glow.x += (glow.targetX - glow.x) * glow.speed;
      glow.y += (glow.targetY - glow.y) * glow.speed;

      // Apply transform: base position + mouse offset + scroll drift
      glow.el.style.transform = `translate(${glow.x}vw, calc(${glow.y}vh + ${scrollInfluence}vh))`;
    });
  });
}
```

### Option B — Canvas Particle Field (Use for tech/SaaS/agency projects)
```javascript
// ═══════════════════════════════════════════════
//  CANVAS PARTICLE BACKGROUND — Interactive field
// ═══════════════════════════════════════════════

/*
  A field of small dots connected by lines when close enough.
  Mouse proximity brightens nearby particles and stretches connections.
  Performance: ~200 particles max. Reduced to 80 on mid-range hardware.

  When to use:
  - Tech, SaaS, developer tools, data products
  - When a "network / connected" metaphor fits the brand
  - NOT for luxury, fashion, or editorial projects

  When NOT to use:
  - Luxury/fashion (too digital)
  - Medical (too technical-feeling)
  - Anything where you need the background completely clean
*/

class ParticleBackground {
  constructor(canvasSelector) {
    this.canvas = document.querySelector(canvasSelector);
    if (!this.canvas) return;

    this.ctx = this.canvas.getContext('2d');
    this.particles = [];
    this.mouse = { x: -9999, y: -9999 };
    this.count = window.innerWidth < 768 ? 0 : 160; // Off on mobile
    this.connectionDistance = 120;
    this.mouseInfluenceRadius = 160;

    this.resize();
    this.createParticles();
    this.bindEvents();
    this.startLoop();
  }

  resize() {
    this.canvas.width  = window.innerWidth;
    this.canvas.height = window.innerHeight;
  }

  createParticles() {
    this.particles = [];
    for (let i = 0; i < this.count; i++) {
      this.particles.push({
        x:   Math.random() * this.canvas.width,
        y:   Math.random() * this.canvas.height,
        vx:  (Math.random() - 0.5) * 0.3,
        vy:  (Math.random() - 0.5) * 0.3,
        size: Math.random() * 1.5 + 0.5,
        alpha: Math.random() * 0.4 + 0.1,
      });
    }
  }

  bindEvents() {
    window.addEventListener('mousemove', (e) => {
      this.mouse.x = e.clientX;
      this.mouse.y = e.clientY;
    }, { passive: true });

    window.addEventListener('resize', () => {
      this.resize();
      this.createParticles();
    });
  }

  draw() {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    // Get accent color from CSS variable
    const accentRaw = getComputedStyle(document.documentElement)
      .getPropertyValue('--accent').trim() || '#6366f1';

    this.particles.forEach((p, i) => {
      // Move
      p.x += p.vx;
      p.y += p.vy;

      // Wrap edges
      if (p.x < 0) p.x = this.canvas.width;
      if (p.x > this.canvas.width) p.x = 0;
      if (p.y < 0) p.y = this.canvas.height;
      if (p.y > this.canvas.height) p.y = 0;

      // Mouse proximity boost
      const dx = this.mouse.x - p.x;
      const dy = this.mouse.y - p.y;
      const dist = Math.sqrt(dx * dx + dy * dy);
      const boost = dist < this.mouseInfluenceRadius
        ? 1 - dist / this.mouseInfluenceRadius
        : 0;

      // Draw particle
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.size + boost * 2, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(255,255,255,${p.alpha + boost * 0.4})`;
      ctx.fill();

      // Draw connections
      for (let j = i + 1; j < this.particles.length; j++) {
        const q = this.particles[j];
        const cx = p.x - q.x;
        const cy = p.y - q.y;
        const cd = Math.sqrt(cx * cx + cy * cy);

        if (cd < this.connectionDistance) {
          const lineAlpha = (1 - cd / this.connectionDistance) * 0.15;
          ctx.beginPath();
          ctx.moveTo(p.x, p.y);
          ctx.lineTo(q.x, q.y);
          ctx.strokeStyle = `rgba(255,255,255,${lineAlpha + boost * 0.1})`;
          ctx.lineWidth = 0.5;
          ctx.stroke();
        }
      }
    });
  }

  startLoop() {
    // Add to GSAP ticker — no separate RAF
    gsap.ticker.add(() => this.draw());
  }
}
```

### JavaScript — Navbar System (Full)
```javascript
// ═══════════════════════════════════════════════
//  NAVBAR SYSTEM — Global, scroll-aware, magnetic
// ═══════════════════════════════════════════════

/*
  BEHAVIORS:
  ─────────────────────────────────────────────
  1. SCROLL STATE MACHINE
     transparent → scrolled: when scroll > navHeight
     scrolled → hidden:     when scrolling DOWN fast (velocity > 2)
     hidden → scrolled:     when scrolling UP at any speed
     This makes the navbar feel responsive without being jumpy.

  2. ACTIVE LINK DETECTION
     Reads current URL path and marks the matching link as active.
     Called on init AND after every Barba page transition.

  3. MAGNETIC LINKS (desktop only)
     Each nav link has a subtle magnetic pull:
     - Mouse enters link bounding box → link text drifts toward cursor
     - Max drift: ±6px X, ±4px Y
     - On mouse leave: spring back to origin
     - Uses GSAP quickTo for performance

  4. MOBILE DRAWER
     Full-screen overlay with staggered link reveal.
     Hamburger icon morphs to X via GSAP.
     Escape key and background click close the drawer.
     Focus trap active when open (accessibility).
*/

function initNavbar() {
  const navbar   = document.getElementById('navbar');
  const hamburger = document.querySelector('.navbar__hamburger');
  const mobileMenu = document.getElementById('mobile-menu');
  if (!navbar) return;

  // ── 1. Scroll State Machine ──────────────────
  const SCROLL_THRESHOLD = parseInt(getComputedStyle(document.documentElement)
    .getPropertyValue('--nav-height')) || 72;

  let lastScrollY = 0;
  let ticking = false;

  function updateNavbarScrollState() {
    const currentScrollY = window.__lenis?.scroll || window.scrollY;
    const scrollingDown  = currentScrollY > lastScrollY;
    const velocity       = Math.abs(currentScrollY - lastScrollY);

    if (currentScrollY <= SCROLL_THRESHOLD) {
      navbar.dataset.state = 'transparent';
    } else if (scrollingDown && velocity > 3) {
      navbar.dataset.state = 'hidden';
    } else if (!scrollingDown) {
      navbar.dataset.state = 'scrolled';
    } else {
      navbar.dataset.state = 'scrolled';
    }

    lastScrollY = currentScrollY;
    ticking = false;
  }

  // Use Lenis scroll event if available, fallback to scroll
  if (window.__lenis) {
    window.__lenis.on('scroll', ({ scroll, velocity }) => {
      if (!ticking) {
        requestAnimationFrame(updateNavbarScrollState);
        ticking = true;
      }
    });
  } else {
    window.addEventListener('scroll', () => {
      if (!ticking) {
        requestAnimationFrame(updateNavbarScrollState);
        ticking = true;
      }
    }, { passive: true });
  }

  // ── 2. Active Link Detection ─────────────────
  updateNavbarForPage();

  // ── 3. Magnetic Links (desktop) ─────────────
  const isMobile = !window.matchMedia('(pointer: fine)').matches;
  if (!isMobile) {
    document.querySelectorAll('.navbar__link').forEach(link => {
      // GSAP quickTo for performance — no event listener overhead
      const xTo = gsap.quickTo(link, 'x', { duration: 0.4, ease: 'power3.out' });
      const yTo = gsap.quickTo(link, 'y', { duration: 0.4, ease: 'power3.out' });

      link.addEventListener('mousemove', (e) => {
        const rect = link.getBoundingClientRect();
        const centerX = rect.left + rect.width  / 2;
        const centerY = rect.top  + rect.height / 2;
        // Drift: max ±6px X, ±4px Y
        xTo((e.clientX - centerX) * 0.4);
        yTo((e.clientY - centerY) * 0.3);
      });

      link.addEventListener('mouseleave', () => {
        xTo(0); yTo(0); // Spring back
      });
    });
  }

  // ── 4. Mobile Drawer ─────────────────────────
  if (hamburger && mobileMenu) {
    let menuOpen = false;

    function openMenu() {
      menuOpen = true;
      hamburger.setAttribute('aria-expanded', 'true');
      mobileMenu.removeAttribute('hidden');
      mobileMenu.style.pointerEvents = 'all';
      document.body.style.overflow = 'hidden'; // Lock scroll

      gsap.timeline()
        .to(mobileMenu, {
          opacity: 1,
          duration: 0.4,
          ease: 'power3.out',
        })
        .to('.mobile-menu__link, .mobile-menu__footer', {
          opacity: 1,
          x: 0,
          stagger: 0.06,
          duration: 0.5,
          ease: 'power3.out',
        }, '-=0.2');
    }

    function closeMenu() {
      menuOpen = false;
      hamburger.setAttribute('aria-expanded', 'false');
      document.body.style.overflow = '';

      gsap.timeline()
        .to('.mobile-menu__link, .mobile-menu__footer', {
          opacity: 0,
          x: -16,
          stagger: { amount: 0.2, from: 'end' },
          duration: 0.3,
          ease: 'power3.in',
        })
        .to(mobileMenu, {
          opacity: 0,
          duration: 0.3,
          ease: 'power3.in',
          onComplete: () => {
            mobileMenu.setAttribute('hidden', '');
            mobileMenu.style.pointerEvents = 'none';
          }
        }, '-=0.1');
    }

    hamburger.addEventListener('click', () => {
      menuOpen ? closeMenu() : openMenu();
    });

    // Close on Escape
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && menuOpen) closeMenu();
    });

    // Close mobile links on click (Barba handles navigation)
    document.querySelectorAll('.mobile-menu__link').forEach(link => {
      link.addEventListener('click', closeMenu);
    });
  }
}

// ═══════════════════════════════════════════════
//  UPDATE NAVBAR FOR PAGE — Called on every nav
// ═══════════════════════════════════════════════
/*
  Must be called:
  1. On initial page load (after initNavbar)
  2. After every Barba page transition (in barba.hooks.after)

  Reads the current pathname and applies aria-current="page"
  and .is-active class to the matching nav link.
*/
function updateNavbarForPage() {
  const currentPath = window.location.pathname;

  document.querySelectorAll('.navbar__link, .mobile-menu__link').forEach(link => {
    const href = link.getAttribute('href');
    const isActive = href && (
      href === currentPath ||
      (href !== '/' && currentPath.startsWith(href))
    );

    link.classList.toggle('is-active', isActive);
    if (isActive) {
      link.setAttribute('aria-current', 'page');
    } else {
      link.removeAttribute('aria-current');
    }
  });
}

// ═══════════════════════════════════════════════
//  NAVBAR ENTRANCE ANIMATION — Runs after intro exit
// ═══════════════════════════════════════════════
/*
  The navbar does NOT appear during the cinematic intro.
  It slides down from above after the CTA is clicked and
  the main experience initializes.

  This creates a clear separation: intro world → site world.
*/
function animateNavbarIn() {
  gsap.from('#navbar', {
    y: -100,
    opacity: 0,
    duration: 0.7,
    ease: 'power3.out',
    delay: 0.2,
  });
}
```

```javascript
// ═══════════════════════════════════════════════
//  THREE.JS ENGINE — Optimized for 60–120fps
// ═══════════════════════════════════════════════
class ThreeEngine {
  constructor(canvasSelector) {
    this.canvas = document.querySelector(canvasSelector);
    this.model = null;
    this.mixer = null;
    this.clock = new THREE.Clock();
    this.mouse = new THREE.Vector2();
    this.targetRotation = new THREE.Vector2();
    this.currentRotation = new THREE.Vector2();
    this.scrollY = 0;
    this.time = 0;
    this.init();
  }

  init() {
    this.setupScene();
    this.setupCamera();
    this.setupRenderer();
    this.setupLights();
    this.setupPostProcessing();
    this.loadModel();
    this.bindEvents();
    this.startRenderLoop();
  }

  setupScene() {
    this.scene = new THREE.Scene();
    this.scene.fog = new THREE.FogExp2(0x050505, 0.08);
  }

  setupCamera() {
    this.camera = new THREE.PerspectiveCamera(
      MODEL_CONFIG.fov,
      window.innerWidth / window.innerHeight,
      0.01, 1000
    );
    const { x, y, z } = MODEL_CONFIG.cameraPosition;
    this.camera.position.set(x, y, z);
  }

  setupRenderer() {
    this.renderer = new THREE.WebGLRenderer({
      canvas: this.canvas,
      alpha: true,
      antialias: true,
      powerPreference: 'high-performance',
      stencil: false,
      depth: true,
    });
    this.renderer.setSize(window.innerWidth, window.innerHeight);
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.toneMappingExposure = 1.2;
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
    this.renderer.shadowMap.enabled = true;
    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;

    this.canvas.style.cssText = `
      position: fixed; top: 0; left: 0;
      width: 100%; height: 100%;
      z-index: 1; pointer-events: none;
    `;
  }

  setupLights() {
    const keyLight = new THREE.DirectionalLight(0xffffff, 2.5);
    keyLight.position.set(3, 5, 3);
    keyLight.castShadow = true;
    keyLight.shadow.mapSize.set(1024, 1024);
    this.scene.add(keyLight);

    const fillLight = new THREE.DirectionalLight(0x4466ff, 0.8);
    fillLight.position.set(-3, 2, -2);
    this.scene.add(fillLight);

    const rimLight = new THREE.DirectionalLight(0xffffff, 1.2);
    rimLight.position.set(0, -2, -4);
    this.scene.add(rimLight);

    const ambient = new THREE.AmbientLight(0xffffff, 0.3);
    this.scene.add(ambient);
  }

  loadModel() {
    const dracoLoader = new THREE.DRACOLoader();
    dracoLoader.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.6/');

    const gltfLoader = new THREE.GLTFLoader();
    gltfLoader.setDRACOLoader(dracoLoader);

    gltfLoader.load(MODEL_CONFIG.path, (gltf) => {
      this.model = gltf.scene;

      if (MODEL_CONFIG.autoCenter) {
        const box = new THREE.Box3().setFromObject(this.model);
        const center = box.getCenter(new THREE.Vector3());
        this.model.position.sub(center);
      }

      if (MODEL_CONFIG.autoScale) {
        const box = new THREE.Box3().setFromObject(this.model);
        const size = box.getSize(new THREE.Vector3());
        const maxDim = Math.max(size.x, size.y, size.z);
        const scale = MODEL_CONFIG.targetSize / maxDim;
        this.model.scale.setScalar(scale);
      }

      this.model.traverse((child) => {
        if (child.isMesh) {
          child.castShadow = true;
          child.receiveShadow = true;
          if (child.material) {
            child.material.envMapIntensity = 1.2;
          }
        }
      });

      if (gltf.animations.length > 0) {
        this.mixer = new THREE.AnimationMixer(this.model);
        const action = this.mixer.clipAction(gltf.animations[0]);
        action.play();
      }

      this.scene.add(this.model);

      // Entrance animation
      gsap.from(this.model.position, { y: this.model.position.y - 1, duration: 1.2, ease: 'power3.out' });
      gsap.from(this.model, { duration: 1.5, ease: 'power3.out',
        onUpdate: () => { this.model.traverse(c => { if (c.isMesh && c.material) c.material.opacity = gsap.getProperty(this.model, 'progress'); }); }
      });
    });
  }

  bindEvents() {
    window.addEventListener('mousemove', (e) => {
      this.mouse.x =  (e.clientX / window.innerWidth  - 0.5) * 2;
      this.mouse.y = -(e.clientY / window.innerHeight - 0.5) * 2;
      this.targetRotation.x = this.mouse.y * MODEL_CONFIG.mouseInfluence;
      this.targetRotation.y = this.mouse.x * MODEL_CONFIG.mouseInfluence;
    }, { passive: true });

    window.__lenis?.on('scroll', ({ scroll }) => { this.scrollY = scroll; });
    window.addEventListener('resize', () => {
      this.camera.aspect = window.innerWidth / window.innerHeight;
      this.camera.updateProjectionMatrix();
      this.renderer.setSize(window.innerWidth, window.innerHeight);
    });
  }

  startRenderLoop() {
    // Add to GSAP ticker — unified RAF loop, no separate requestAnimationFrame
    gsap.ticker.add((time) => {
      this.time = time;
      const delta = this.clock.getDelta();

      if (this.mixer) this.mixer.update(delta);

      if (this.model) {
        // Lerp rotation toward target
        this.currentRotation.x += (this.targetRotation.x - this.currentRotation.x) * 0.08;
        this.currentRotation.y += (this.targetRotation.y - this.currentRotation.y) * 0.08;

        // Idle float
        this.model.position.y = Math.sin(time * MODEL_CONFIG.idleFloatSpeed) * MODEL_CONFIG.idleFloatAmplitude;

        // Apply rotations
        this.model.rotation.x = this.currentRotation.x;
        this.model.rotation.y += MODEL_CONFIG.idleRotationSpeed + this.currentRotation.y;

        // Scroll influence
        const scrollProgress = this.scrollY / document.body.scrollHeight;
        this.model.rotation.y += scrollProgress * MODEL_CONFIG.scrollRotationFactor * 0.01;
        this.model.position.z  = scrollProgress * MODEL_CONFIG.scrollPositionFactor * -1;
      }

      this.renderer.render(this.scene, this.camera);
    });
  }

  setupPostProcessing() {
    // Post-processing setup placeholder
    // Add: UnrealBloomPass for glow, SSAOPass for ambient occlusion
    // Only activate on high-performance tier (AdaptiveQuality check)
  }
}
```

```javascript
        this.visible = true;
        gsap.to([this.dot, this.ring], { opacity: 1, duration: 0.3 });
      }
    }, { passive: true });

    // Hover detection — change cursor state for interactive elements
    document.querySelectorAll('a, button, [data-cursor="hover"]').forEach(el => {
      el.addEventListener('mouseenter', () => this.setState('hover'));
      el.addEventListener('mouseleave', () => this.setState('default'));
    });

    document.querySelectorAll('p, span, h1, h2, h3, [data-cursor="text"]').forEach(el => {
      el.addEventListener('mouseenter', () => this.setState('text'));
      el.addEventListener('mouseleave', () => this.setState('default'));
    });

    document.addEventListener('mouseleave', () => {
      gsap.to([this.dot, this.ring], { opacity: 0, duration: 0.3 });
      this.visible = false;
    });
  }

  setState(state) {
    if (this.state === state) return;
    this.state = state;

    const states = {
      default: { dot: { scale: 1,    opacity: 1 }, ring: { scale: 1,    opacity: 1,   borderColor: 'rgba(255,255,255,0.5)' } },
      hover:   { dot: { scale: 0.4,  opacity: 0 }, ring: { scale: 2.2,  opacity: 0.8, borderColor: 'var(--accent)' } },
      text:    { dot: { scale: 6,    opacity: 0.15, borderRadius: '2px' }, ring: { scale: 0,    opacity: 0 } },
    };

    const s = states[state] || states.default;
    gsap.to(this.dot,  { ...s.dot,  duration: 0.35, ease: 'power3.out' });
    gsap.to(this.ring, { ...s.ring, duration: 0.5,  ease: 'power3.out' });
  }

  startLoop() {
    gsap.ticker.add(() => {
      // Dot snaps instantly
      gsap.set(this.dot, { x: this.pos.x, y: this.pos.y });

      // Ring lags behind — feels physical
      this.ring_pos.x += (this.pos.x - this.ring_pos.x) * 0.15;
      this.ring_pos.y += (this.pos.y - this.ring_pos.y) * 0.15;
      gsap.set(this.ring, { x: this.ring_pos.x, y: this.ring_pos.y });
    });
  }
}
```

## ⚡ PERFORMANCE — ADAPTIVE QUALITY SYSTEM

```javascript
// ═══════════════════════════════════════════════
//  ADAPTIVE QUALITY — FPS-based degradation
// ═══════════════════════════════════════════════

/*
  Monitors real FPS continuously.
  If FPS drops below threshold → downgrades visual quality automatically.
  
  Tiers:
  HIGH:   Full effects. Particles + shaders + post-processing + full DPR
  MEDIUM: Reduced particles. No post-processing. DPR capped at 1.5.
  LOW:    Particles off. Shaders simplified. DPR = 1. Shadows off.

  This system is invisible to the user — quality degrades gracefully
  without any UI indication.
*/
class AdaptiveQuality {
  constructor(renderer) {
    this.renderer = renderer;
    this.tier = 'HIGH';
    this.frameCount = 0;
    this.lastTime = performance.now();
    this.fps = 60;
    this.history = [];

    gsap.ticker.add(() => this.measure());
  }

  measure() {
    this.frameCount++;
    const now = performance.now();

    if (now - this.lastTime >= 1000) {
      this.fps = this.frameCount;
      this.frameCount = 0;
      this.lastTime = now;
      this.history.push(this.fps);
      if (this.history.length > 5) this.history.shift();
      this.evaluate();
    }
  }

  evaluate() {
    const avg = this.history.reduce((a, b) => a + b, 0) / this.history.length;

    if (avg < 30 && this.tier !== 'LOW') {
      this.setTier('LOW');
    } else if (avg < 45 && this.tier === 'HIGH') {
      this.setTier('MEDIUM');
    } else if (avg > 55 && this.tier === 'LOW') {
      this.setTier('MEDIUM');
    } else if (avg > 58 && this.tier === 'MEDIUM') {
      this.setTier('HIGH');
    }
  }

  setTier(tier) {
    if (this.tier === tier) return;
    this.tier = tier;
    console.log(`[AdaptiveQuality] → ${tier} (${Math.round(this.history.at(-1))}fps)`);

    const configs = {
      HIGH:   { dpr: Math.min(window.devicePixelRatio, 2), shadows: true },
      MEDIUM: { dpr: Math.min(window.devicePixelRatio, 1.5), shadows: true },
      LOW:    { dpr: 1, shadows: false },
    };

    const config = configs[tier];
    this.renderer.setPixelRatio(config.dpr);
    this.renderer.shadowMap.enabled = config.shadows;

    document.documentElement.dataset.quality = tier.toLowerCase();
  }
}

// CSS hooks for quality tier:
// [data-quality="low"]    .particles { display: none; }
// [data-quality="medium"] .shader-bg { opacity: 0.5; }
```
