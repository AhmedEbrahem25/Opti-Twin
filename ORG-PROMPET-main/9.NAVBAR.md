# MODULE: NAVBAR SYSTEM
**ID:** 08 | **Load order:** 8 | **Depends on:** ANIMATION, SCROLL, TOKENS | **Required:** always

---

## RESPONSIBILITY
Global persistent navbar: scroll-state machine, active link detection, magnetic hover,  
mobile hamburger + full-screen drawer, entrance animation, and Barba-aware page updates.

## SCOPE BOUNDARIES
- ✅ Navbar scroll states (transparent / scrolled / hidden)
- ✅ Active link marking (`updateNavbarForPage`)
- ✅ Magnetic link effect (desktop only)
- ✅ Mobile hamburger + drawer (GSAP animated)
- ✅ Focus trap and keyboard accessibility
- ✅ Navbar entrance animation (post-intro)
- ❌ Does NOT manage page transitions (→ SCROLL module)
- ❌ Does NOT manage route content (→ Barba view hooks in SCROLL module)

## WHEN TO USE
Every project with navigation.

## WHEN NOT TO USE
Single-page apps with anchor links only — use a simplified version without Barba hooks.

## CONFLICTS
- `initNavbar()` must be called INSIDE `initMainExperience()` — never during intro
- `updateNavbarForPage()` must be called after EVERY Barba transition
- Lenis `stop()`/`start()` must wrap mobile menu open/close

## JAVASCRIPT

```javascript
// File: scripts/navbar.js

function initNavbar() {
  const navbar   = document.getElementById('navbar');
  if (!navbar) return;

  // ── 1. SCROLL STATE MACHINE ───────────────────
  let lastScroll = 0;
  const navHeight = parseFloat(getComputedStyle(document.documentElement)
    .getPropertyValue('--nav-height'));

  window.__lenis?.on('scroll', ({ scroll, velocity }) => {
    const direction = scroll > lastScroll ? 'down' : 'up';
    lastScroll = scroll;

    if (scroll <= 10) {
      navbar.dataset.state = 'transparent';
    } else if (direction === 'down' && Math.abs(velocity) > 2 && scroll > navHeight * 2) {
      navbar.dataset.state = 'hidden';
    } else {
      navbar.dataset.state = 'scrolled';
    }
  });

  // ── 2. ACTIVE LINK ────────────────────────────
  updateNavbarForPage();

  // ── 3. MAGNETIC LINKS (desktop only) ─────────
  if (window.matchMedia('(pointer: fine)').matches) {
    initMagneticNav();
  }

  // ── 4. MOBILE DRAWER ─────────────────────────
  initMobileMenu();
}

function initMagneticNav() {
  document.querySelectorAll('.navbar__link').forEach(link => {
    link.addEventListener('mousemove', (e) => {
      const rect = link.getBoundingClientRect();
      const x = e.clientX - (rect.left + rect.width  / 2);
      const y = e.clientY - (rect.top  + rect.height / 2);
      gsap.to(link, { x: x * 0.25, y: y * 0.25, duration: 0.3, ease: 'power2.out' });
    });
    link.addEventListener('mouseleave', () => {
      gsap.to(link, { x: 0, y: 0, duration: 0.5, ease: 'elastic.out(1, 0.4)' });
    });
  });
}

function initMobileMenu() {
  const hamburger  = document.querySelector('.navbar__hamburger');
  const mobileMenu = document.getElementById('mobile-menu');
  if (!hamburger || !mobileMenu) return;

  let menuOpen = false;

  function openMenu() {
    menuOpen = true;
    mobileMenu.hidden = false;
    mobileMenu.style.pointerEvents = 'auto';
    hamburger.setAttribute('aria-expanded', 'true');
    hamburger.setAttribute('aria-label', 'Close menu');
    window.__lenis?.stop();

    gsap.timeline()
      .to(mobileMenu, { opacity: 1, duration: 0.4, ease: 'power3.out' })
      .to('.mobile-menu__link, .mobile-menu__footer', {
        opacity: 1, x: 0, stagger: 0.07,
        duration: 0.5, ease: 'power3.out',
      }, '-=0.2');
  }

  function closeMenu() {
    menuOpen = false;
    hamburger.setAttribute('aria-expanded', 'false');
    hamburger.setAttribute('aria-label', 'Open menu');
    window.__lenis?.start();

    gsap.timeline()
      .to('.mobile-menu__link, .mobile-menu__footer', {
        opacity: 0, x: -16, stagger: 0.04,
        duration: 0.3, ease: 'power3.in',
      })
      .to(mobileMenu, {
        opacity: 0, duration: 0.3, ease: 'power3.in',
        onComplete: () => {
          mobileMenu.hidden = true;
          mobileMenu.style.pointerEvents = 'none';
        },
      }, '-=0.1');
  }

  hamburger.addEventListener('click', () => menuOpen ? closeMenu() : openMenu());
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape' && menuOpen) closeMenu(); });
  document.querySelectorAll('.mobile-menu__link').forEach(l => l.addEventListener('click', closeMenu));
}

// Called on init AND after every Barba transition
function updateNavbarForPage() {
  const path = window.location.pathname;
  document.querySelectorAll('.navbar__link, .mobile-menu__link').forEach(link => {
    const href = link.getAttribute('href');
    const active = href && (href === path || (href !== '/' && path.startsWith(href)));
    link.classList.toggle('is-active', active);
    active
      ? link.setAttribute('aria-current', 'page')
      : link.removeAttribute('aria-current');
  });
}

// Entrance animation — called from initMainExperience()
function animateNavbarIn() {
  gsap.from('#navbar', { y: -100, opacity: 0, duration: 0.7, ease: 'power3.out', delay: 0.2 });
}
```

## PERFORMANCE IMPACT
- **Low** — magnetic effect uses `gsap.to()` on `mousemove`. Throttle if needed on slow devices.
- Mobile drawer uses display+opacity — no layout reflows.
- Scroll state machine uses Lenis scroll event — already in the GSAP ticker.

## IMPLEMENTATION NOTES
- File: `scripts/navbar.js`
- `initNavbar()` fires inside `initMainExperience()` — never before
- `updateNavbarForPage()` must be in `barba.hooks.after` in SCROLL module
- Logo should always be inline SVG — enables color transitions via CSS `currentColor`
- Never use `<img>` for logo — cannot transition color smoothly
